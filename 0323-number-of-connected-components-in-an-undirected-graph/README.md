## Explanation

Iterate through every vertex. Whenever you find an unvisited vertex, increment the component count and start BFS/DFS from it.

---

## My Notes

A simpler and cleaner approach:
We have n nodes. We can simply iterate through every node.
Whenever we encounter an unvisited node, we have discovered a new connected component.
We start BFS from that node and mark all reachable nodes as visited.

class Solution {
public:
    int countComponents(int n, vector<vector<int>>& edges) {

        vector<vector<int>> adj(n);

        for (auto& e : edges) {
            int u = e[0];
            int v = e[1];

            adj[u].push_back(v);
            adj[v].push_back(u);
        }

        vector<bool> visited(n, false);

        int count = 0;

        for (int i = 0; i < n; i++) {

            // Found a new connected component
            if (!visited[i]) {

                count++;

                queue<int> q;
                q.push(i);
                visited[i] = true;

                while (!q.empty()) {

                    int node = q.front();
                    q.pop();

                    for (int neighbour : adj[node]) {

                        if (!visited[neighbour]) {

                            visited[neighbour] = true;
                            q.push(neighbour);
                        }
                    }
                }
            }
        }

        return count;
    }
};
